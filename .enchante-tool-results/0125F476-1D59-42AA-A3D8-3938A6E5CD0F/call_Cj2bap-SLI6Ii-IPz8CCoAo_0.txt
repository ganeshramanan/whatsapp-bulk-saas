 396	      const payload = {
 397	        businessName: document.getElementById('regBusinessName').value.trim(),
 398	        email: document.getElementById('regEmail').value.trim(),
 399	        password: document.getElementById('regPassword').value,
 400	        phoneNumberId: document.getElementById('regPhoneId').value.trim(),
 401	        accessToken: document.getElementById('regToken').value.trim(),

 405	        const res = await fetch('/api/auth/register', {
 406	          method: 'POST',
 407	          headers: { 'Content-Type': 'application/json' },
 408	          body: JSON.stringify(payload),
 409	        });
 410	        const data = await res.json();
 411	        if (res.ok) {
 412	          localStorage.setItem('grambi_token', data.token);
 413	          showDashboard(data.user);
 414	        } else {
 415	          // Extract specific error message
 416	          let errMsg = 'Registration failed';
 417	          if (typeof data.error === 'string') {
 418	            errMsg = data.error;
 419	          } else if (data.error && typeof data.error === 'object') {
 420	            const keys = Object.keys(data.error);
 421	            if (keys.length > 0 && data.error[keys[0]]?._errors) {
 422	              errMsg = `${keys[0]}: ${data.error[keys[0]]._errors.join(', ')}`;
 423	            }
 424	          }
 425	          showAuthAlert(errMsg, 'error');
 426	        }
 427	      } catch (err) {
 428	        showAuthAlert('Network error: ' + (err.message || 'Cannot reach server'), 'error');
 429	      }
 430	    });
